# CErXes syntax cases, run by tests/syntax.rs (cargo test) and by
# bench/cases.mjs on the Ranger build (rgrc src/lib.rs, JavaScript).
#
# A case: a line `==== <syntax>` (tsx, ts, jsx or js), the script, a line
# `----`, and the value the script answers (its last expression statement,
# or its error). Lines starting with `#` before a case are comments.

# types are erased: annotations, predicates, `this` parameters
==== tsx
let a: number = 1; const b: Array<Map<string, number[]>> = []; var c!: string; a + b.length
----
1

==== tsx
function f(a: number, b?: string, ...r: number[]): string { return a + (b ?? '') + r.length; } f(1, 'x', 3, 4)
----
1x2

==== tsx
const f = (x: number, y = 2): number => x * y; f(3)
----
6

==== tsx
const f = async (x: number): Promise<number> => x; typeof f
----
function

==== tsx
const g = ({ a, b }: { a: number; b: number }): number => a + b; g({ a: 1, b: 2 })
----
3

==== tsx
try { throw 1 } catch (e: unknown) { String(e) }
----
1

==== tsx
let t: [string, number?, ...boolean[]] = ['a']; let u: (a: string) => void = () => {}; let k: keyof typeof t = 0; t[k]
----
a

==== tsx
type F = <T>(x: T) => T extends string ? 'yes' : 'no'; let o: { readonly [k: string]: number } = { x: 1 }; o.x
----
1

==== tsx
function isStr(x: unknown): x is string { return typeof x === 'string'; } isStr('a')
----
true

==== tsx
function check(x: unknown): asserts x is number {} check(1); 'ok'
----
ok

==== tsx
function f(this: Window, a: number) { return a; } f(4)
----
4

==== tsx
let x: `id-${number}` | -1 | typeof globalThis | import('x').Y = -1; x
----
-1

# classes: modifiers, parameter properties, abstract members, overloads; decorators are an error
==== tsx

        abstract class Shape<T = {}> implements Named {
            static count: number = 0;
            private readonly id: number;
            declare tag: string;
            [key: string]: any;
            protected constructor(public readonly name: string, private sides?: number) { this.id = ++Shape.count; }
            abstract area(): number;
            describe(): string { return `${this.name}#${this.id}:${this.area()}`; }
            get label(): string { return this.name; }
            opt?(): void;
        }
        class Sq extends Shape<{}> {
            override area(): number { return this.s * this.s; }
            constructor(private s: number) { super('sq', 4); }
            static of<T>(n: number): Sq { return new Sq(n); }
        }
        const q = Sq.of<number>(3);
        [q.describe(), q.label, q.s, 'tag' in q, 'opt' in q, q.sides].join()
----
sq#1:9,sq,3,false,false,4

# JSX components: functions, members, classes, null
==== tsx

        const Hello = ({ name, children }) => <b>Hi {name}{children}</b>;
        function Wrap(props) { return <section>{props.children}</section>; }
        const UI = { Title: (p) => <h1>{p.text}</h1> };
        class Counter { constructor(p) { this.n = p.start; } render() { return <i>{this.n}</i>; } }
